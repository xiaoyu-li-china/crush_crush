## 3. 激励视频：发奖判定

```ts
/** 比流量主后台设置的最长观看时长（此处为 15 秒）略短 */
const REWARDED_MIN_WATCH_MS = 14_000;

function isRewardedVideoCompleted(
  res: { isEnded?: boolean } | null | undefined,
  shownMs: number,
): boolean {
  if (shownMs >= REWARDED_MIN_WATCH_MS) return true;
  if (!res || typeof res.isEnded !== 'boolean') return true;
  return res.isEnded;
}
```

## 4. 激励视频：展示

要点：每次展示只注册一次 `onClose` 并在结束时注销；`show()` 失败先 `load()` 再试；加一个超时兜底，保证 Promise 一定会结束，界面不会永远卡在「加载中」。

```ts
function showRewarded(ad: WxRewardedVideoAd): Promise<AdShowResult> {
  const startedAt = Date.now();

  return new Promise((resolve) => {
    let settled = false;
    let timer = 0;

    const finish = (result: AdShowResult) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      ad.offClose(onClose);
      resolve(result);
      void ad.load().catch(() => {}); // 预加载下一条
    };

    const onClose = (res?: { isEnded?: boolean }) => {
      finish(isRewardedVideoCompleted(res, Date.now() - startedAt) ? 'completed' : 'skipped');
    };

    ad.onClose(onClose);

    // 一直没能弹出来就结束等待；弹出成功后放宽到 3 分钟等关闭
    timer = setTimeout(() => finish('not_ready'), 12_000) as unknown as number;

    ad.show()
      .catch(() => ad.load().then(() => ad.show()))
      .then(() => {
        clearTimeout(timer);
        timer = setTimeout(() => finish('skipped'), 180_000) as unknown as number;
      })
      .catch(() => finish('not_ready'));
  });
}
```

## 5. 插屏：频控判断（纯函数，便于单元测试）

```ts
function shouldShowInterstitial(args: {
  levelId: number;
  won: boolean;
  lastShownMs: number;
  nowMs: number;
  everyNLevels: number;
  cooldownSeconds: number;
}): boolean {
  const { levelId, won, lastShownMs, nowMs, everyNLevels, cooldownSeconds } = args;
  if (!won || everyNLevels <= 0) return false;
  if (levelId % everyNLevels !== 0) return false;
  if (lastShownMs > 0 && nowMs - lastShownMs < cooldownSeconds * 1000) return false;
  return true;
}
```

## 6. 启动插屏：处理 2001 频控

```ts
async function tryLaunchInterstitial(opts: {
  ads: IAdService;
  stillInLobby: () => boolean;
  delayMs: number;    // 示例项目 2500
  retryMs: number;    // 示例项目 2500
  timeoutMs: number;  // 示例项目 45000
}): Promise<void> {
  const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
  const deadline = Date.now() + opts.timeoutMs;

  await sleep(opts.delayMs); // 先让大厅画出来
  while (Date.now() < deadline && opts.stillInLobby()) {
    const result = await opts.ads.show('interstitial_settle');
    if (result === 'completed') return;
    await sleep(opts.retryMs); // 多数失败是 2001，隔一会再试
  }
}
```

## 7. 大厅横幅：防止迟到出现在关卡里

用一个「现在还想不想显示」的标记，在加载完成和 `show()` 返回时都再检查一次。

```ts
let bannerWanted = false;

async function showLobbyBanner(banner: WxCustomAd) {
  bannerWanted = true;
  try {
    await banner.show();
    if (!bannerWanted) banner.hide(); // 显示成功时玩家已经进关了
  } catch {
    // 无填充等，忽略即可
  }
}

function hideLobbyBanner(banner: WxCustomAd) {
  bannerWanted = false;
  try { banner.hide(); } catch { /* 模拟器里可能报 removeImageView not found */ }
}

banner.onLoad(() => {
  if (!bannerWanted) hideLobbyBanner(banner);
});
```

## 8. 前台闸门：统一处理广告与前后台

把第 06 章第 3 节的规则落成一个类。`pause` / `resume` 由你的游戏循环提供。

```ts
const HIDE_DEBOUNCE_MS = 120;

class ForegroundGate {
  private hidden = false;
  private overlayDepth = 0;
  private hideTimer = 0;
  private ignoreHideUntil = 0;

  constructor(
    private readonly pause: () => void,
    private readonly resume: (reason: 'show' | 'overlay-end' | 'overlay-abort') => void,
  ) {}

  /** 横幅刚创建时短暂忽略 onHide，避免大厅黑屏 */
  holdNativeChrome(ms = 1600) {
    this.ignoreHideUntil = Math.max(this.ignoreHideUntil, Date.now() + ms);
  }

  /** 打开游戏圈等原生页：点击当下就停，避免「微信无响应」 */
  forcePause() {
    this.cancelHideTimer();
    if (this.overlayDepth > 0 || this.hidden) return;
    this.hidden = true;
    this.pause();
  }

  /** 激励 / 插屏展示前调用 */
  enterOverlay() {
    this.overlayDepth += 1;
    this.cancelHideTimer();
    if (!this.hidden) {
      this.hidden = true;
      this.pause();
    }
  }

  /** 广告正常关闭 */
  leaveOverlay() { this.finishOverlay('overlay-end'); }

  /** show() 立刻失败：直接恢复，不做画布重建 */
  abortOverlay() { this.finishOverlay('overlay-abort'); }

  onHide() {
    if (this.overlayDepth > 0 || this.hidden) return;
    if (Date.now() < this.ignoreHideUntil) return;
    this.cancelHideTimer();
    this.hideTimer = setTimeout(() => {
      this.hideTimer = 0;
      if (this.overlayDepth > 0 || this.hidden) return;
      this.hidden = true;
      this.pause();
    }, HIDE_DEBOUNCE_MS) as unknown as number;
  }

  onShow() {
    const hadPendingHide = this.hideTimer !== 0;
    this.cancelHideTimer();
    if (this.overlayDepth > 0) return;
    if (!this.hidden && !hadPendingHide) return;
    this.hidden = false;
    this.resume('show');
  }

  private finishOverlay(reason: 'overlay-end' | 'overlay-abort') {
    this.overlayDepth = Math.max(0, this.overlayDepth - 1);
    if (this.overlayDepth > 0) return;
    this.hidden = false;
    this.resume(reason);
  }

  private cancelHideTimer() {
    if (!this.hideTimer) return;
    clearTimeout(this.hideTimer);
    this.hideTimer = 0;
  }
}
```

接入方式：

```ts
const gate = new ForegroundGate(pauseLoop, onResume);
wx.onHide(() => gate.onHide());
wx.onShow(() => gate.onShow());

async function watchRewarded(placement: AdPlacement): Promise<AdShowResult> {
  const startedAt = Date.now();
  gate.enterOverlay();
  const result = await ads.show(placement);
  // 不到 80 毫秒就返回，说明广告根本没弹出来
  if (Date.now() - startedAt < 80) gate.abortOverlay();
  else gate.leaveOverlay();
  return result;
}
```

## 9. 广告关闭后的恢复

```ts
function onResume(reason: 'show' | 'overlay-end' | 'overlay-abort') {
  const isDevtools = wx.getSystemInfoSync().platform === 'devtools';
  if (reason === 'overlay-abort') {
    kickLoop();
    return;
  }
  const settleMs = isDevtools ? 50 : 480;
  const muteInputMs = isDevtools ? 160 : 800;

  muteInputFor(muteInputMs); // 防止关闭广告的点击穿透到棋盘
  setTimeout(() => {
    // 开发者工具里重设 canvas.width 会卡死，跳过
    if (!isDevtools && canvas.width !== expectedCanvasWidth()) {
      restoreCanvasSurface();
    }
    if (bgmStarted && !muted) resumeBgm();
    kickLoop();
  }, settleMs);
}
```

`kickLoop`、`muteInputFor`、`restoreCanvasSurface`、`resumeBgm` 需要你按自己的渲染和音频实现提供。
